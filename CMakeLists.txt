cmake_minimum_required(VERSION 3.8)
project(ultrasound_usb)

find_package(ament_cmake REQUIRED)
find_package(Threads REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(image_transport REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio highgui)

add_executable(ultrasound_usb_node src/ultrasound_usb_node.cpp)
target_compile_features(ultrasound_usb_node PUBLIC cxx_std_17)
target_compile_options(ultrasound_usb_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(ultrasound_usb_node rclcpp sensor_msgs std_msgs rcl_interfaces cv_bridge image_transport)
target_include_directories(ultrasound_usb_node PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(ultrasound_usb_node ${OpenCV_LIBRARIES} Threads::Threads)

install(TARGETS ultrasound_usb_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(device_path_test tests/device_path_test.cpp)
  target_compile_features(device_path_test PRIVATE cxx_std_17)
  target_compile_options(device_path_test PRIVATE -Wall -Wextra -Wpedantic)
  target_include_directories(device_path_test PRIVATE src)
  add_test(NAME device_path_reconnect COMMAND device_path_test)
endif()

ament_package()
